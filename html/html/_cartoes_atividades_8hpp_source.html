532 void mover(
533 const Identificador &identificador,
534 const Estado &novoEstado,
535 const Timestamp &momento)
536 {
537
538 CartaoAtividade cartao = consultar(identificador);
539
540 std::string estadoAtual = cartao.getEstado().getEstado();
541 std::string estadoDestino = novoEstado.getEstado();
542
543 if (estadoAtual == "A FAZER" && estadoDestino == "FAZENDO")
544 {
545
546 cartao.setEstado(novoEstado);
547 cartao.setInicio(momento);
548 }
549 else if (
550 estadoAtual == "FAZENDO" && estadoDestino == "FEITO")
551 {
552
553 cartao.setEstado(novoEstado);
554 cartao.setTermino(momento);
555 }
556 else
557 {
558
559 throw std::invalid_argument(
560 "Movimentacao nao permitida.");
561 }
562
563 atualizar(cartao);
