49 void validarLimiteWIP(
50 const CartaoAtividade &cartao,
51 const std::string &ignorarIdentificador = "")
52 {
53 Quadro quadro = repositorioQuadros.consultar(cartao.getQuadro());
54 int limite = quadro.getLimite().getValor();
55 int quantidade = 0;
56
57 std::vector<CartaoAtividade> cartoes = repositorio.listar();
58
59 for (const CartaoAtividade &existente : cartoes)
60 {
61 bool mesmoQuadro =
62 existente.getQuadro().getIdentificador() ==
63 cartao.getQuadro().getIdentificador();
64
65 bool estaFazendo =
66 existente.getEstado().getEstado() == "FAZENDO";
67
68 bool naoIgnorar =
69 existente.getIdentificador().getIdentificador() !=
70 ignorarIdentificador;
71
72 if (mesmoQuadro && estaFazendo && naoIgnorar)
73 {
74 quantidade++;
75 }
76 }
77
78 if (quantidade >= limite)
79 {
80 throw std::runtime_error("Limite de WIP do quadro atingido.");
81 }
82 }

91 long long converterParaMinutos(const Timestamp &timestamp)
92 {
93 std::string valor = timestamp.getTimestamp();
94
95 int dia = std::stoi(valor.substr(0, 2));
96 std::string mes = valor.substr(3, 3);
97 int ano = std::stoi(valor.substr(7, 4));
98 int hora = std::stoi(valor.substr(12, 2));
99 int minuto = std::stoi(valor.substr(15, 2));
100
101 int numeroMes;
102
103 if (mes == "JAN")
104 numeroMes = 0;
105 else if (mes == "FEV")
106 numeroMes = 1;
107 else if (mes == "MAR")
108 numeroMes = 2;
109 else if (mes == "ABR")
110 numeroMes = 3;
111 else if (mes == "MAI")
112 numeroMes = 4;
113 else if (mes == "JUN")
114 numeroMes = 5;
115 else if (mes == "JUL")
116 numeroMes = 6;
117 else if (mes == "AGO")
118 numeroMes = 7;
119 else if (mes == "SET")
120 numeroMes = 8;
121 else if (mes == "OUT")
122 numeroMes = 9;
123 else if (mes == "NOV")
124 numeroMes = 10;
125 else if (mes == "DEZ")
126 numeroMes = 11;
127 else
128 {
129 throw std::invalid_argument("Mes invalido.");
130 }
131
132 std::tm data = {};
133 data.tm_mday = dia;
134 data.tm_mon = numeroMes;
135 data.tm_year = ano - 1900;
136 data.tm_hour = hora;
137 data.tm_min = minuto;
138 data.tm_isdst = -1;
139
140 std::time_t resultado = std::mktime(&data);
141
142 if (resultado == static_cast<std::time_t>(-1))
143 {
144 throw std::runtime_error("Nao foi possivel converter a data.");
145 }
146
147 return static_cast<long long>(resultado / 60);
148 }

159 double calcularTempoMedio(
160 const Identificador &quadro,
161 bool ciclo)
162 {
163 std::vector<CartaoAtividade> cartoes = repositorio.listar();
164
165 long long soma = 0;
166 int quantidade = 0;
167
168 for (const CartaoAtividade &cartao : cartoes)
169 {
170 bool mesmoQuadro =
171 cartao.getQuadro().getIdentificador() ==
172 quadro.getIdentificador();
173
174 bool concluido =
175 cartao.getEstado().getEstado() == "FEITO";
176
177 if (!mesmoQuadro || !concluido)
178 {
179 continue;
180 }
181
182 Timestamp inicio;
183
184 if (ciclo)
185 {
186 inicio = cartao.getInicio();
187 }
188 else
189 {
190 inicio = cartao.getEntrada();
191 }
192
193 long long momentoInicial = converterParaMinutos(inicio);
194 long long momentoFinal =
195 converterParaMinutos(cartao.getTermino());
196
197 if (momentoFinal < momentoInicial)
198 {
199 throw std::runtime_error(
200 "Data de conclusao anterior a data inicial.");
201 }
202
203 soma += momentoFinal - momentoInicial;
204 quantidade++;
205 }
206
207 if (quantidade == 0)
208 {
209 return 0.0;
210 }
211
212 return static_cast<double>(soma) / quantidade;
213 }
